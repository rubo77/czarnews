### Make table 'cn_words'
CREATE TABLE `cn_words` (
`id` BIGINT(20) NOT NULL AUTO_INCREMENT PRIMARY KEY, 
`word` VARCHAR(255) NOT NULL, 
`type` VARCHAR(255) NOT NULL, 
`replace` LONGTEXT NOT NULL
); 

### Make table 'cn_comments'
CREATE TABLE `cn_comments` (
  `id` bigint(20) NOT NULL default '0',
  `news_id` bigint(20) NOT NULL default '0',
  `name` varchar(255) NOT NULL default '',
  `email` varchar(255) NOT NULL default '',
  `comment` text NOT NULL,
  `date` varchar(10) NOT NULL default ''
);

### Added 'words' column to user table (for user admin access to 'words')
ALTER TABLE `cn_users` ADD `words` ENUM('on', 'off') NOT NULL;

### Make 'user' field UNIQUE
ALTER TABLE `cn_users` ADD UNIQUE (`user`) ;

### Add 'source' and 'version' column to configuration table
ALTER TABLE `cn_config` ADD `source` VARCHAR( 255 ) NOT NULL ;
ALTER TABLE `cn_config` ADD `author` VARCHAR( 255 ) NOT NULL; 
ALTER TABLE `cn_config` ADD `version` VARCHAR( 255 ) NOT NULL ;

### Add columns to CONFIG table
ALTER TABLE `cn_config` ADD `words` ENUM( 'on', 'off' ) NOT NULL ;
ALTER TABLE `cn_config` ADD `comments` ENUM( 'on', 'off' ) NOT NULL ;
ALTER TABLE `cn_config` ADD `search` ENUM( 'on', 'off' ) NOT NULL ;
ALTER TABLE `cn_config` ADD `pages` ENUM( 'on', 'off' ) NOT NULL ;
ALTER TABLE `cn_config` ADD `catbox` ENUM( 'on', 'off' ) NOT NULL ; 

### Add 'content2' column to news table
ALTER TABLE `cn_news` ADD `content2` LONGTEXT NOT NULL AFTER `content`; 
ALTER TABLE `cn_news` ADD `sumstory` ENUM('on','off') DEFAULT 'off' NOT NULL AFTER `content2`;
